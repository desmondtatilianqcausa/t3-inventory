Create an analytics utility to track events:

```typescript
// lib/analytics.ts
export const trackEvent = (eventName, properties = {}) => {
  // Implementation depends on your analytics provider
  // Example with Google Analytics
  if (typeof window !== 'undefined' && window.gtag) {
    window.gtag('event', eventName, properties);
  }
  
  // Log for development
  if (process.env.NODE_ENV === 'development') {
    console.log(`[Analytics] ${eventName}`, properties);
  }
};

export const trackPageView = (url, eventData = {}) => {
  trackEvent('page_view', {
    page_path: url,
    ...eventData
  });
};
```

Add a layout effect to track page views:

```tsx
'use client';
import { useEffect } from 'react';
import { usePathname } from 'next/navigation';
import { trackPageView } from '@/lib/analytics';

export function AnalyticsProvider({ children, eventId, eventSlug }) {
  const pathname = usePathname();
  
  useEffect(() => {
    // Track page view
    trackPageView(pathname, { eventId, eventSlug });
  }, [pathname, eventId, eventSlug]);
  
  return children;
}
```
