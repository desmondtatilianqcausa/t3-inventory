Add JSON-LD structured data for the event:
```typescript
function EventJsonLd({ event }) {
  const jsonLd = {
    '@context': 'https://schema.org',
    '@type': 'Event',
    name: event.title,
    startDate: new Date(event.startAt).toISOString(),
    endDate: new Date(event.endAt).toISOString(),
    location: {
      '@type': 'Place',
      name: event.venue,
      address: {
        '@type': 'PostalAddress',
        streetAddress: event.address,
        addressLocality: event.city,
        addressRegion: event.state,
      },
    },
    image: event.images[0],
    description: event.description,
    url: `https://yourdomain.com/event/${event.slug}`,
    offers: event.ticketUrl ? {
      '@type': 'Offer',
      url: event.ticketUrl,
      price: event.price,
      priceCurrency: 'USD',
      availability: event.isSoldOut ? 'SoldOut' : 'InStock',
    } : undefined,
  };

  return (
    <script
      type="application/ld+json"
      dangerouslySetInnerHTML={{ __html: JSON.stringify(jsonLd) }}
    />
  );
}
```
