Create a server route to generate ICS files dynamically:

```typescript
// app/api/calendar/[eventId]/route.ts
import { NextResponse } from 'next/server';
import { getEvent } from '@/lib/convex';
import { createEvent } from 'ics';

export async function GET(request, { params }) {
  const event = await getEvent(params.eventId);
  if (!event) {
    return NextResponse.json({ error: 'Event not found' }, { status: 404 });
  }
  
  const startDate = new Date(event.startAt);
  const endDate = new Date(event.endAt);
  
  const icsEvent = {
    start: [startDate.getFullYear(), startDate.getMonth() + 1, startDate.getDate(), startDate.getHours(), startDate.getMinutes()],
    end: [endDate.getFullYear(), endDate.getMonth() + 1, endDate.getDate(), endDate.getHours(), endDate.getMinutes()],
    title: event.title,
    description: event.description,
    location: `${event.venue}, ${event.address}, ${event.city}, ${event.state}`,
    url: `https://yourdomain.com/event/${event.slug}`,
    status: 'CONFIRMED',
  };
  
  return new Promise((resolve) => {
    createEvent(icsEvent, (error, value) => {
      if (error) {
        resolve(NextResponse.json({ error: 'Failed to create ICS file' }, { status: 500 }));
        return;
      }
      
      resolve(new NextResponse(value, {
        headers: {
          'Content-Type': 'text/calendar',
          'Content-Disposition': `attachment; filename="${event.slug}.ics"`,
        },
      }));
    });
  });
}
```
