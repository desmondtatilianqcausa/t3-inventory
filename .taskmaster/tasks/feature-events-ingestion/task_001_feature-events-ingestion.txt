Example endpoint implementation:
```python
@app.post('/scrape/events')
async def scrape_events(request: EventScrapeRequest):
    try:
        # Initialize scraper with config
        scraper = EventScraper(config)
        
        # Run scraping job
        results = await scraper.scrape(request.url, request.cityId, request.city, request.state)
        
        # Return results
        return {
            'status': 'success',
            'jobId': results.job_id,
            'candidates': results.candidates
        }
    except Exception as e:
        logger.error(f"Scraping error: {str(e)}")
        return {
            'status': 'error',
            'message': str(e)
        }
```
