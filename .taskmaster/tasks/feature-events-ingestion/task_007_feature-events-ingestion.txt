# Details:
Create a React component for viewing and managing import jobs:

1. Table/list view of jobs with status indicators
2. Filtering and sorting options
3. Job details expansion with logs
4. Actions (retry, cancel, view candidates)

Example implementation:
```tsx
// components/admin/EventImportJobs.tsx
import { useState } from 'react';
import { useQuery, useMutation } from 'convex/react';
import { api } from '../../convex/_generated/api';
import { formatDistanceToNow } from 'date-fns';

export default function EventImportJobs() {
  const [expandedJobId, setExpandedJobId] = useState(null);
  const [filter, setFilter] = useState('all'); // all, pending, completed, failed
  
  const jobs = useQuery(api.eventImportJobs.list) || [];
  const retryJob = useMutation(api.actions.events.retryImportJob);
  
  // Filter jobs based on status
  const filteredJobs = filter === 'all' 
    ? jobs 
    : jobs.filter(job => job.status === filter);
  
  // Sort by most recent first
  const sortedJobs = [...filteredJobs].sort((a, b) => b.startedAt - a.startedAt);
  
  const handleRetry = async (jobId) => {
    try {
      await retryJob({ jobId });
    } catch (err) {
      console.error('Failed to retry job:', err);
    }
  };
  
  const toggleExpand = (jobId) => {
    setExpandedJobId(expandedJobId === jobId ? null : jobId);
  };
  
  return (
    <div className="p-4 bg-white rounded shadow">
      <div className="flex justify-between items-center mb-4">
        <h2 className="text-xl font-bold">Event Import Jobs</h2>
        
        <div className="flex space-x-2">
          <select
            value={filter}
            onChange={(e) => setFilter(e.target.value)}
            className="p-2 border rounded"
          >
            <option value="all">All Jobs</option>
            <option value="pending">Pending</option>
            <option value="running">Running</option>
            <option value="completed">Completed</option>
            <option value="failed">Failed</option>
          </select>
        </div>
      </div>
      
      {sortedJobs.length === 0 ? (
        <p className="text-gray-500">No jobs found.</p>
      ) : (
        <div className="overflow-x-auto">
          <table className="min-w-full divide-y divide-gray-200">
            <thead>
              <tr>
                <th className="px-4 py-2 text-left">Source</th>
                <th className="px-4 py-2 text-left">Status</th>
                <th className="px-4 py-2 text-left">Started</th>
                <th className="px-4 py-2 text-left">Candidates</th>
                <th className="px-4 py-2 text-left">Actions</th>
              </tr>
            </thead>
            <tbody className="divide-y divide-gray-200">
              {sortedJobs.map((job) => (
                <>
                  <tr key={job._id} className="hover:bg-gray-50">
                    <td className="px-4 py-2">
                      {job.source?.url || 'Unknown'}
                    </td>
                    <td className="px-4 py-2">
                      <span className={`px-2 py-1 rounded text-xs ${getStatusColor(job.status)}`}>
                        {job.status}
                      </span>
                    </td>
                    <td className="px-4 py-2">
                      {formatDistanceToNow(new Date(job.startedAt), { addSuffix: true })}
                    </td>
                    <td className="px-4 py-2">
                      {job.candidatesCount}
                    </td>
                    <td className="px-4 py-2 space-x-2">
                      <button
                        onClick={() => toggleExpand(job._id)}
                        className="px-2 py-1 text-xs bg-gray-200 rounded"
                      >
                        {expandedJobId === job._id ? 'Hide' : 'Logs'}
                      </button>
                      
                      {job.status === 'failed' && (
                        <button
                          onClick={() => handleRetry(job._id)}
                          className="px-2 py-1 text-xs bg-blue-500 text-white rounded"
                        >
                          Retry
                        </button>
                      )}
                      
                      {job.candidatesCount > 0 && (
                        <a
                          href={`/admin/events/candidates?jobId=${job._id}`}
                          className="px-2 py-1 text-xs bg-green-500 text-white rounded"
                        >
                          Review
                        </a>
                      )}
                    </td>
                  </tr>
                  
                  {expandedJobId === job._id && (
                    <tr>
                      <td colSpan={5} className="px-4 py-2 bg-gray-50">
                        <div className="max-h-40 overflow-y-auto">
                          {job.logs.map((log, i) => (
                            <div key={i} className="text-sm mb-1">
                              <span className="text-gray-500">
                                {new Date(log.timestamp).toLocaleTimeString()}
                              </span>
                              <span className={`ml-2 ${getLogLevelColor(log.level)}`}>
                                [{log.level}]
                              </span>
                              <span className="ml-2">{log.message}</span>
                            </div>
                          ))}
                          
                          {job.error && (
                            <div className="text-sm text-red-500 mt-2">
                              Error: {job.error}
                            </div>
                          )}
                        </div>
                      </td>
                    </tr>
                  )}
                </>
              ))}
            </tbody>
          </table>
        </div>
      )}
    </div>
  );
}

function getStatusColor(status) {
  switch (status) {
    case 'pending': return 'bg-yellow-100 text-yellow-800';
    case 'running': return 'bg-blue-100 text-blue-800';
    case 'completed': return 'bg-green-100 text-green-800';
    case 'failed': return 'bg-red-100 text-red-800';
    default: return 'bg-gray-100 text-gray-800';
  }
}

function getLogLevelColor(level) {
  switch (level.toLowerCase()) {
    case 'error': return 'text-red-600';
    case 'warn': return 'text-yellow-600';
    case 'info': return 'text-blue-600';
    default: return 'text-gray-600';
  }
}
```
