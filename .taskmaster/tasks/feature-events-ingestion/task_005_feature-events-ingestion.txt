// utils/eventNormalizer.ts
export function normalizeTitle(title: string): string {
  return title
    .toLowerCase()
    .replace(/[^\w\s]/g, '') // Remove special chars
    .replace(/\s+/g, ' ')    // Normalize whitespace
    .trim();
}

export function normalizeVenue(venue: string): string {
  return venue
    .toLowerCase()
    .replace(/[^\w\s]/g, '')
    .replace(/\s+/g, ' ')
    .trim();
}

export function normalizeDate(date: Date): string {
  return date.toISOString().split('T')[0]; // YYYY-MM-DD
}

export function generateDedupeKey(title: string, venue: string, date: Date): string {
  return [
    normalizeTitle(title),
    normalizeVenue(venue || ''),
    normalizeDate(date)
  ].join('|');
}

export function calculateConfidence(candidate: EventCandidate): number {
  let score = 0;
  let fields = 0;
  
  // Title is required
  score += candidate.title ? 1 : 0;
  fields += 1;
  
  // Optional fields
  if (candidate.startDate) { score += 1; fields += 1; }
  if (candidate.venue) { score += 1; fields += 1; }
  if (candidate.description) { score += 0.5; fields += 0.5; }
  if (candidate.imageUrl) { score += 0.5; fields += 0.5; }
  
  // Calculate percentage
  return fields > 0 ? (score / fields) : 0;
}
